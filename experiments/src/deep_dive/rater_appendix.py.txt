

# ============================================================================= LLM item-rating offset (deep dive 2026-10-26)
# Appended to the xp_level_m050 model.py by deep_dive/build_rater.py, ONLY if the pre-registered gate passed
# (experiments/results/deep_dive/items/eval_items_*.json).  Adds  w * (s1 - mu) / sd + b  to the prior offset of every
# item (labelled entries and targets) when its rating is available, where s1 = log P(Yes) - log P(No) of the
# pre-registered prompt under RATER_MODEL_ID.  Everything else is the m050 predictor.  Safety:
#   * at most RATER_SLOTS worker processes hold the model at once (O_EXCL slot files with the holder's pid; a slot whose
#     pid is gone is reclaimed); a worker without a slot uses only ratings already in the shared disk cache;
#   * the model is loaded in a background thread (never inside a scoring call's critical path) with a hard cap;
#   * ratings are computed only for items needed by the current call, at most RATER_MAX_PER_CALL per call, and never
#     after RATER_WALL_BUDGET seconds since the first worker of the run started;
#   * any failure (import, CUDA, OOM, timeout, tokenizer, I/O) disables the rater in that process -> plain m050;
#   * skipped entirely under the organisers' local smoke check (PREDICTIVE_EVAL_LOCAL_SMOKE_TEST=1) and when
#     PAIEC_RATER_DISABLE=1.
RATER_MODEL_ID = "Qwen/Qwen3-VL-4B-Instruct"
RATER_W = __RATER_W__
RATER_B = __RATER_B__
RATER_MU = __RATER_MU__
RATER_SD = __RATER_SD__
RATER_SLOTS = 4
RATER_LOAD_CAP = 300.0
RATER_MAX_PER_CALL = 400
RATER_WALL_BUDGET = 4.5 * 3600.0
RATER_CONTENT_CHARS, RATER_FEATURE_CHARS = 1500, 300
RATER_PROMPT = ("You are estimating how hard a benchmark task is for state-of-the-art AI systems released in 2025-2026 (for "
                "example GPT-5, Claude Opus 4, Gemini 2.5 Pro), possibly running inside an agent harness. Read the task and "
                "answer with one word: Yes if such a system would most likely complete it correctly, No otherwise.\n\n"
                "Task:\n<<<\n{content}\n>>>\n\nItem attributes: {features}\n\n"
                "Would a state-of-the-art 2025-2026 AI system most likely complete this task correctly? Answer Yes or No.")
_RATER_SKIP = (os.environ.get("PREDICTIVE_EVAL_LOCAL_SMOKE_TEST") == "1" or os.environ.get("PAIEC_RATER_DISABLE") == "1")
_R = {"state": "init", "model": None, "tok": None, "yes": None, "no": None, "cache": {}, "cache_mtime": {}, "dir": None,
      "slot": None, "thread": None, "t0": None}
_R_LOCK = _threading.RLock()


def _rater_dir():
    if _R["dir"] is None:
        import tempfile
        for d in (os.path.join(os.path.dirname(os.path.abspath(__file__)), ".paiec_rater"),
                  os.path.join(tempfile.gettempdir(), "paiec_rater")):
            try:
                os.makedirs(d, exist_ok=True)
                probe = os.path.join(d, ".w%d" % os.getpid())
                with open(probe, "w") as fh:
                    fh.write("1")
                os.remove(probe)
                _R["dir"] = d
                break
            except Exception:
                continue
        if _R["dir"] is None:
            _R["dir"] = ""
    return _R["dir"]


def _rater_prompt(item):
    c = " ".join(_s(item.get("item_content")).split())[:RATER_CONTENT_CHARS]
    f = " ".join(_s(item.get("item_features")).split())[:RATER_FEATURE_CHARS] or "none"
    return RATER_PROMPT.format(content=c, features=f)


def _rater_key(prompt):
    import hashlib as _hl
    return _hl.sha1(prompt.encode("utf-8")).hexdigest()


def _rater_run_start():
    """Wall-clock start of the run (first worker that got here), shared through a file."""
    if _R["t0"] is None:
        import time as _t
        d = _rater_dir()
        t0 = _t.time()
        if d:
            p = os.path.join(d, "run_start")
            try:
                fd = os.open(p, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
                os.write(fd, repr(t0).encode())
                os.close(fd)
            except FileExistsError:
                try:
                    t0 = float(open(p).read().strip())
                except Exception:
                    pass
            except Exception:
                pass
        _R["t0"] = t0
    return _R["t0"]


def _rater_reload_cache():
    import time as _t
    now = _t.time()
    if now - _R.get("reload_t", 0.0) < 2.0:
        return
    _R["reload_t"] = now
    d = _rater_dir()
    if not d:
        return
    try:
        names = [n for n in os.listdir(d) if n.startswith("ratings_") and n.endswith(".jsonl")]
    except Exception:
        return
    for n in names:
        p = os.path.join(d, n)
        try:
            m = os.path.getmtime(p)
            if _R["cache_mtime"].get(n) == m:
                continue
            with open(p, encoding="utf-8") as fh:
                for line in fh:
                    try:
                        k, v = line.rstrip("\n").split("\t")
                        _R["cache"][k] = float(v)
                    except Exception:
                        continue
            _R["cache_mtime"][n] = m
        except Exception:
            continue


def _pid_alive(pid):
    try:
        os.kill(pid, 0)
        return True
    except ProcessLookupError:
        return False
    except Exception:
        return True


def _rater_take_slot():
    d = _rater_dir()
    if not d:
        return None
    for k in range(RATER_SLOTS):
        p = os.path.join(d, "slot_%d" % k)
        for _attempt in range(2):
            try:
                fd = os.open(p, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
                os.write(fd, str(os.getpid()).encode())
                os.close(fd)
                return p
            except FileExistsError:
                try:
                    pid = int(open(p).read().strip() or "0")
                except Exception:
                    pid = 0
                if pid and not _pid_alive(pid):
                    try:
                        os.remove(p)
                        continue
                    except Exception:
                        pass
                break
            except Exception:
                return None
    return None


def _rater_load():
    try:
        import torch
        from transformers import AutoModelForImageTextToText, AutoTokenizer
        if not torch.cuda.is_available():
            raise RuntimeError("no cuda")
        free, _total = torch.cuda.mem_get_info(0)
        if free < 12 * 2 ** 30:
            raise RuntimeError("gpu memory")
        tok = AutoTokenizer.from_pretrained(RATER_MODEL_ID, local_files_only=True)
        try:
            m = AutoModelForImageTextToText.from_pretrained(RATER_MODEL_ID, dtype=torch.bfloat16, local_files_only=True)
        except TypeError:
            m = AutoModelForImageTextToText.from_pretrained(RATER_MODEL_ID, torch_dtype=torch.bfloat16, local_files_only=True)
        m = m.to("cuda:0").eval()
        ids = lambda s: tok.encode(s, add_special_tokens=False)
        yes = sorted({ids(v)[0] for v in ("Yes", " Yes") if ids(v)})
        no = sorted({ids(v)[0] for v in ("No", " No") if ids(v)})
        if not yes or not no:
            raise RuntimeError("tokens")
        _R.update(model=m, tok=tok, yes=yes, no=no, torch=torch, state="ready")
    except Exception:
        _R["state"] = "failed"
        _rater_release_slot()


def _rater_release_slot():
    p = _R.get("slot")
    if p:
        try:
            os.remove(p)
        except Exception:
            pass
        _R["slot"] = None


def _rater_start():
    """Called on the first rating request in this process: take a slot and start the background load."""
    if _R["state"] != "init":
        return
    _R["state"] = "noslot"
    _rater_run_start()
    p = _rater_take_slot()
    if p is None:
        return
    _R["slot"] = p
    import atexit
    atexit.register(_rater_release_slot)
    _R["state"] = "loading"
    th = _threading.Thread(target=_rater_load, daemon=True)
    _R["thread"] = th
    th.start()


def _rater_score(prompt):
    torch, tok, m = _R["torch"], _R["tok"], _R["model"]
    msgs = [{"role": "user", "content": prompt}]
    try:
        txt = tok.apply_chat_template(msgs, tokenize=False, add_generation_prompt=True, enable_thinking=False)
    except TypeError:
        txt = tok.apply_chat_template(msgs, tokenize=False, add_generation_prompt=True)
    enc = tok(txt, return_tensors="pt", add_special_tokens=False).to("cuda:0")
    with torch.no_grad():
        try:
            out = m(**enc, logits_to_keep=1)
        except TypeError:
            out = m(**enc)
    last = torch.log_softmax(out.logits[0, -1].float(), dim=-1)
    s1 = (torch.logsumexp(last[_R["yes"]], 0) - torch.logsumexp(last[_R["no"]], 0)).item()
    if not math.isfinite(s1):
        raise ValueError("non-finite")
    return float(s1)


def rater_offsets(items):
    """Item offsets (logits) for a list of item dicts; 0.0 where no rating is available."""
    if _RATER_SKIP or not items:
        return [0.0] * len(items)
    try:
        with _R_LOCK:
            prompts = [_rater_prompt(it) if isinstance(it, dict) else "" for it in items]
            keys = [_rater_key(p) if p else "" for p in prompts]
            _rater_start()
            _rater_reload_cache()
            missing = [i for i, k in enumerate(keys) if k and k not in _R["cache"]]
            if missing and _R["state"] == "loading" and _R["thread"] is not None:
                _R["thread"].join(0.0)
            import time as _t
            if (missing and _R["state"] == "ready"
                    and _t.time() - _rater_run_start() < RATER_WALL_BUDGET):
                d = _rater_dir()
                new = []
                seen = set()
                for i in missing[:RATER_MAX_PER_CALL]:
                    k = keys[i]
                    if k in seen or k in _R["cache"]:
                        continue
                    seen.add(k)
                    try:
                        v = _rater_score(prompts[i])
                    except Exception:
                        _R["state"] = "failed"
                        break
                    _R["cache"][k] = v
                    new.append((k, v))
                if new and d:
                    try:
                        with open(os.path.join(d, "ratings_%d.jsonl" % os.getpid()), "a", encoding="utf-8") as fh:
                            for k, v in new:
                                fh.write("%s\t%r\n" % (k, v))
                    except Exception:
                        pass
            # consistent level: once the rater is active in this process (model ready or any cached rating), an item
            # without a rating gets the average offset b (z = 0), so labelled entries and targets share one level;
            # with the rater inactive every offset is 0 (plain m050)
            active = _R["state"] == "ready" or bool(_R["cache"])
            out = []
            for k in keys:
                v = _R["cache"].get(k) if k else None
                if v is not None:
                    out.append(RATER_W * (v - RATER_MU) / RATER_SD + RATER_B)
                else:
                    out.append(RATER_B if active else 0.0)
            return out
    except Exception:
        return [0.0] * len(items)


def rater_epoch():
    """Changes whenever the set of available ratings or the rater state changes (part of the XP cache key)."""
    if _RATER_SKIP:
        return (0, "skip")
    return (len(_R["cache"]), _R["state"])


# ---- hooks into the m050 predictor (offsets of labelled entries and targets; budget-0 prior)
_core_xp_state = _xp_state
_core_xp_clear_cache = xp_clear_cache


def xp_clear_cache() -> None:  # noqa: F811
    _core_xp_clear_cache()
    _XP_CACHE["rater_entry"] = None


_RATER_EKEYS: dict = {}


def _xp_state(labeled):  # noqa: F811
    """m050's cross-pair state with rating offsets on the labelled entries.  The entries' rating keys are computed once
    per labelled list; the fit is redone only when the number of rated entries or the rater's activity changes."""
    if _RATER_SKIP:
        return _core_xp_state(labeled)
    fp = _xp_fingerprint(labeled)
    ek = _RATER_EKEYS.get(fp)
    if ek is None:
        items = []
        for entry in labeled:
            try:
                it = entry[0][1]
            except Exception:
                it = None
            items.append(it if isinstance(it, dict) else {})
        ek = (items, [_rater_key(_rater_prompt(it)) if it else "" for it in items])
        if len(_RATER_EKEYS) > 8:
            _RATER_EKEYS.clear()
        _RATER_EKEYS[fp] = ek
    items, keys = ek
    try:
        with _R_LOCK:
            _rater_start()
            _rater_reload_cache()
            cache = _R["cache"]
            n_rated = sum(1 for k in keys if k and k in cache)
            active = _R["state"] == "ready" or bool(cache)
            can_rate = _R["state"] == "ready" and n_rated < sum(1 for k in keys if k)
    except Exception:
        n_rated, active, can_rate = 0, False, False
    key = ("rater", fp, n_rated, active)
    hit = _XP_CACHE.get("rater_entry")
    if hit is not None and hit[0] == key and not can_rate:
        return hit[1]
    offs = rater_offsets(items) if active or can_rate else [0.0] * len(items)
    with _XP_LOCK:
        ks, ys, offv, iks = [], [], [], []
        s_cache: dict = {}
        for entry, ro in zip(labeled, offs):
            try:
                pair, label = entry[0], entry[1]
                subj, item = pair[0], pair[1]
                if not isinstance(subj, dict) or not isinstance(item, dict):
                    continue
                y = _label_value(label)
                if y is None:
                    continue
                skey = xp_subject_key(subj)
                zs = s_cache.get(skey)
                if zs is None:
                    zs = _subject_logit(subj)
                    s_cache[skey] = zs
                ik = xp_item_key(item)
                ks.append((ik[0], _s(subj.get("normalized_name")).strip().lower(), skey))
                ys.append(float(y))
                offv.append(min(max(zs + _item_logit(item), -PRIOR_LOGIT_CAP), PRIOR_LOGIT_CAP) + ro)
                iks.append(ik)
            except Exception:
                continue
        state = xp_build(ks, ys, offv, iks) if ys else None
        try:
            with _R_LOCK:
                n_after = sum(1 for k in keys if k and k in _R["cache"])
                act_after = _R["state"] == "ready" or bool(_R["cache"])
        except Exception:
            n_after, act_after = n_rated, active
        _XP_CACHE["rater_entry"] = (("rater", fp, n_after, act_after), state)
        return state


def predict_cross_pair(input, labeled):  # noqa: F811
    if not labeled or len(labeled) > XP_MAX_ENTRIES:
        return None
    subject, item = input[0], input[1]
    if not isinstance(subject, dict) or not isinstance(item, dict):
        return None
    state = _xp_state(labeled)
    if state is None:
        return None
    ik = xp_item_key(item)
    ro = rater_offsets([item])[0]
    off = min(max(_subject_logit(subject) + _item_logit(item), -PRIOR_LOGIT_CAP), PRIOR_LOGIT_CAP) + ro
    p = _xp_predict_keys(state, ik[0], _s(subject.get("normalized_name")).strip().lower(), xp_subject_key(subject), ik, off)
    return _clip(p)


_core_predict = predict


def predict(input, labeled=None):  # noqa: F811
    if not _RATER_SKIP and not labeled:
        try:
            subject, item = input[0], input[1]
            if isinstance(subject, dict) and isinstance(item, dict):
                ro = rater_offsets([item])[0]
                if ro != 0.0:
                    z = min(max(_subject_logit(subject) + _item_logit(item), -PRIOR_LOGIT_CAP), PRIOR_LOGIT_CAP) + ro
                    return _clip(_sigmoid(z))
        except Exception:
            pass
    return _core_predict(input, labeled)
